"""CPU-only abstract interleavings; not a firmware or hardware simulator.
Only tests whether a completed background apply can explain saved R125B fields.
Actual scheduling, atomicity, interrupts and runtime identity are not proven.
"""
import json
base=["force_A", "read_gate", "store_target", "direct_apply", "sample"]
def run(events):
 s=dict(A=0,B=0,target=0,old=0,logical_applied=0);trace=[];gate=None
 for e in events:
  if e=="force_A":s["A"]=1
  elif e=="read_gate":gate=[s["A"],s["B"]]
  elif e=="store_target":s["target"]=1250
  elif e in ("direct_apply","background_apply") and s["A"]!=s["B"]:
   s["old"]=s["target"];s["logical_applied"]=s["target"];s["A"]=s["B"]
  trace.append(dict(event=e,state=dict(s)))
 return dict(events=events,gate_read=gate,final=s,matches_saved_fields=gate==[1,0] and s["A"]==s["B"]==0 and s["target"]==1250 and s["old"]==0,trace=trace)
cases=[run(base)]+[run(base[:i]+["background_apply"]+base[i:]) for i in range(len(base))]
# Distinguishes a lost selector transition from callback absence alone.
assert not cases[0]["matches_saved_fields"]
assert [x["events"].index("background_apply") for x in cases[1:] if x["matches_saved_fields"]]==[2]
print(json.dumps(dict(classification="CONSISTENT_WITH only",scope="one entry, whole completed apply events, no resets",cases=cases),indent=2))

// Decode only a caller-selected interval in a disposable, readOnly session.
// No device access, image writes, or persistent project modification.
public class BC250BoundedDecode extends GhidraScript {
 public void run() throws Exception {
  String[] args=getScriptArgs(); Address start=toAddr(Long.decode(args[1]));
  Address end=toAddr(Long.decode(args[2])); AddressSet bounds=new AddressSet(start,end);
  try(PrintWriter out=new PrintWriter(args[0])) {
   out.println("EXECUTABLE_SHA256="+currentProgram.getExecutableSHA256());
   out.println("BOUNDED_STATIC_DECODE="+start+".."+end);
   DisassembleCommand cmd=new DisassembleCommand(start,bounds,true);
   out.println("DISASSEMBLE_RESULT="+cmd.applyTo(currentProgram,monitor));
   InstructionIterator it=currentProgram.getListing().getInstructions(bounds,true);
   while(it.hasNext()) {
    Instruction ins=it.next();out.print("INS="+ins.getAddress()+" ");
    for(byte b:ins.getBytes())out.printf("%02x",b&255);
    out.println(" "+ins);
    for(Reference r:ins.getReferencesFrom())out.println("REF="+r);
   }
   out.println("NO_FUNCTION_BOUNDARY_CREATED=true");
  }
 }
